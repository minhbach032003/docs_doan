|Store Owner|
start
:Create Stock Count Plan;
:Select warehouse,\nlocation and scope;

|System|
:Create Stock Count Session;
:Capture book stock snapshot;

|Inventory Staff|
:Perform physical count;
:Enter actual quantity\nby product, lot and location;

|System|
:Compare actual quantity\nwith book quantity;
:Calculate variance;

if (Variance found?) then (Yes)

    |Inventory Staff|
    :Review variance;

    if (Recount required?) then (Yes)

        :Perform recount;
        :Update actual quantity;

        |System|
        :Recalculate variance;

    else (No)

        |Inventory Staff|
        :Keep current count result;

    endif

    |Inventory Staff|
    :Enter variance reason;
    :Submit count result;

    |System|
    :Record count result\nand variance details;

    |Store Owner|
    :Review count result;

    if (Approve adjustment?) then (Yes)

        |System|
        :Create approved\nInventory Adjustment;

        |Inventory Staff|
        :Confirm Inventory Adjustment;

        |System|
        :Update stock by warehouse,\nlot and location;
        :Record adjustment history;
        :Close Stock Count Session;

    else (No)

        |System|
        :Return result for recount;

        |Inventory Staff|
        :Review rejected result;
        :Perform recount;
        :Update actual quantity;
        :Resubmit count result;

        |System|
        :Recalculate variance;
        :Record revised count result;

    endif

else (No)

    |System|
    :Confirm no variance;
    :Close Stock Count Session\nwithout adjustment;

endif

stop
@enduml
