    if (Accept partial receipt?) then (Yes)
        |System|
        :Create partial transfer receipt;
        :Keep unresolved quantity under exception handling;
    else (No)
        |System|
        :Reject affected quantity;
        :Create return-back handling\nfor unresolved quantity;
    endif
